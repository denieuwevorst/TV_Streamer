
[stream]

; Uitvoermodus:
; web    = alleen browserstream (HLS)
; stream = alleen MPEG-TS (VLC/IPTV)
; both   = beide tegelijkertijd
output_mode = both

; Map met videobestanden
folder = ./videos

; Netwerkinstellingen
host = 0.0.0.0
port = 8080

; Videoresolutie
width = 1280
height = 720

; Videokwaliteit
video_bitrate = 2500k
audio_bitrate = 128k
fps = 25

; FFmpeg encoder preset
; ultrafast, veryfast, faster, fast, medium
preset = veryfast

; GPU-encoding:
; none = CPU (libx264)
; qsv  = Intel Quick Sync (h264_qsv)
; vaapi = Intel/AMD via VAAPI (h264_vaapi, Linux; als qsv niet werkt)
hwaccel = none

; Bij vaapi: ook decoderen, schalen en logo op de GPU (true/false)
; Zet op false als dit fouten geeft; dan alleen encoderen op de GPU.
gpu_filters = false

; QSV preset: veryfast, faster, fast, medium, slow, slower, veryslow
qsv_preset = veryfast

; Linux render-device voor QSV (leeg laten op Windows)
qsv_device = /dev/dri/renderD128

; HLS-instellingen
hls_time = 4
hls_list_size = 8
hls_dir = hls

; Shuffle instellen
shuffle = true

[logo]
enabled = true
file = ./logo.png
width = 160
margin = 20

[web]
page_title = DTV
# none = alleen de vaste titel
# filename = actuele bestandsnaam toevoegen
